+
+        Reject candidates whose velocity direction points toward (rather than
+        away from) nearby obstacles. This prevents counterflow patterns that

+        Args:

+            return emergency_mask

+        if ("v_cmd" not in self.action_spec.names

+        if "theta" in self.state_spec.names:
+            theta = float(state[self.state_spec.index("theta")])
+        else:
+            theta = 0.0

+        validated_mask = emergency_mask.copy()
+        rejected_count = 0

+        # Check each emergency candidate
+        for k in np.where(emergency_mask)[0]:
+            # Get first-step control (emergency patterns are prefix-only)
+            v_cmd = float(samples[k, 0, v_index])
+            omega_cmd = float(samples[k, 0, omega_index])
+
+            # Skip stationary candidates (always safe)
+            if abs(v_cmd) < 1e-6:
+                continue
+
+            # Compute velocity direction in world frame
+            # For differential drive: heading_rate = omega, forward_speed = v
+            motion_heading = theta  # First-step heading (omega effect is small)
+            vel_world = np.array([

+                v_cmd * np.sin(motion_heading)

+            vel_norm = np.linalg.norm(vel_world)
+            if vel_norm < 1e-6:

+    def _compute_barrier_action(
+        self,
+        state,
+        probabilistic_obstacles,
+        stop_max_probability,
+        stop_probability_mass,
+    ):
+        """
+        Compute barrier action when all candidates violate hard threshold.
+
+        Priority:
+        1. Emergency stop (if safe)
+        2. Gentle deceleration (preserve maneuverability)
+        3. None (fall back to stop)
+
+        Args:
+            state: Current robot state
+            probabilistic_obstacles: List of tracked obstacles
+            stop_max_probability: Risk of stopping
+            stop_probability_mass: Accumulated mass of stopping
+
+        Returns:
+            (action, reason): Barrier action and reason string, or (None, None)
+        """
+        # Check if we have velocity control
+        if "v_cmd" not in self.action_spec.names:
+            return None, None
+
+        v_index = self.action_spec.index("v_cmd")
+        omega_index = (
+            self.action_spec.index("omega_cmd")
+            if "omega_cmd" in self.action_spec.names
+            else None
+        )
+
+        # Get current velocity
+        if "v" in self.state_spec.names:
+            current_v = float(state[self.state_spec.index("v")])
+        else:
+            current_v = 0.0
+
+        # Priority 1: Emergency stop if collision risk is acceptable
+        emergency_stop_feasible = (
+            stop_max_probability
+            < self.config.probabilistic_obstacle_hard_probability_threshold
+        )
+
+        if emergency_stop_feasible:
+            action = np.zeros(self.action_spec.dimension, dtype=np.float64)
+            return action, "emergency_stop"
+
+        # Priority 2: Gentle deceleration
+        # Decelerate but preserve turning capability
+        decel_rate = 0.3  # m/s per step
+        target_v = max(0.0, current_v - decel_rate * self.config.dt)
+
+        # Steer toward maximum clearance direction if obstacles present
+        target_omega = 0.0
+        if (
+            probabilistic_obstacles
+            and omega_index is not None
+            and "theta" in self.state_spec.names
+        ):
+            # Find direction with maximum clearance
+            robot_pos = np.array([
+                state[self.state_spec.position_indices[0]],
+                state[self.state_spec.position_indices[1]]
+            ], dtype=np.float64)
+            theta = float(state[self.state_spec.index("theta")])
+
+            # Sample 8 directions
+            best_clearance = -1.0
+            best_heading = theta
+            for angle_offset in np.linspace(-np.pi, np.pi, 9)[:-1]:
+                test_heading = theta + angle_offset
+                min_clearance_in_direction = float('inf')
+
+                for obs in probabilistic_obstacles:

+                    )
+                    # Only consider obstacles ahead

+                                np.sin(test_heading)
+                            ])
+                        )
+                        min_clearance_in_direction = min(
+                            min_clearance_in_direction, lateral_dist

+                np.sin(best_heading - theta),
+                np.cos(best_heading - theta)
+            )
+            target_omega = 2.0 * heading_error  # P gain = 2.0

+        if omega_index is not None:
+            action[omega_index] = target_omega
+
+        # Clip to action bounds
+        action = self.action_spec.clip(action)
+
+        return action, "gentle_deceleration"
